import type { MiddlewareHandler } from '../../types'
import { MemoryStore, type Store } from './store'
/** 滑动窗口：上一窗口按剩余比例计入，边界不再突刺。Sliding window: the last one counted by the share left. */
export const rateLimiter = (options: {
limit: number
windowMs: number
store?: Store
}): MiddlewareHandler => {
const store = options.store ?? new MemoryStore()
return async (c, next) => {
const key = c.req.header('x-forwarded-for') ?? 'anonymous'
const { current, previous, elapsed } = await store.hit(key, options.windowMs)
const weight = 1 - elapsed / options.windowMs
const count = previous * weight + current
c.header('RateLimit', `limit=${options.limit}, remaining=${Math.max(0, options.limit - count)}`)
if (count > options.limit) {
return c.text('Too Many Requests', 429)
}
await next()
}
